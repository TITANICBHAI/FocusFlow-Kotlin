/**
 * Branded splash shown while FocusFlow completes its boot sequence.
 *
 * Animation is intentionally implemented natively in Compose rather than
 * using a video/GIF:
 *
 *  1. Purple background is immediately visible.
 *  2. Shield + translucent circular halo scale/fade into place.
 *  3. FocusFlow title slides/fades upward.
 *  4. Tagline appears with the title.
 *  5. Completed frame is held until application boot is ready.
 */
@Composable
fun FocusFlowSplashOverlay(
    visible: Boolean,
    modifier: Modifier = Modifier,
) {
    var logoVisible by rememberSaveable { mutableStateOf(false) }
    var textVisible by rememberSaveable { mutableStateOf(false) }
    var sequenceComplete by rememberSaveable { mutableStateOf(false) }

    LaunchedEffect(Unit) {
        if (!logoVisible) {
            logoVisible = true
            kotlinx.coroutines.delay(LOGO_DURATION_MS.toLong())
        }

        if (!textVisible) {
            textVisible = true
            kotlinx.coroutines.delay(TEXT_DURATION_MS.toLong())
        }

        if (!sequenceComplete) {
            kotlinx.coroutines.delay(HOLD_AFTER_TEXT_MS)
            sequenceComplete = true
        }
    }

    AnimatedVisibility(
        visible = visible || !sequenceComplete,
        enter = fadeIn(
            animationSpec = tween(160),
        ),
        exit = fadeOut(
            animationSpec = tween(220),
        ),
        modifier = modifier,
    ) {
        Box(
            modifier = Modifier
                .fillMaxSize()
                .background(BrandPrimary),
            contentAlignment = Alignment.Center,
        ) {
            Column(
                horizontalAlignment = Alignment.CenterHorizontally,
            ) {
                AnimatedVisibility(
                    visible = logoVisible,
                    enter = fadeIn(
                        animationSpec = tween(LOGO_DURATION_MS),
                    ) + scaleIn(
                        initialScale = 0.72f,
                        animationSpec = tween(LOGO_DURATION_MS),
                    ),
                ) {
                    FocusFlowSplashLogo()
                }

                Spacer(
                    modifier = Modifier.height(24.dp),
                )

                AnimatedVisibility(
                    visible = textVisible,
                    enter = fadeIn(
                        animationSpec = tween(TEXT_DURATION_MS),
                    ) + slideInVertically(
                        initialOffsetY = { 22 },
                        animationSpec = tween(TEXT_DURATION_MS),
                    ),
                ) {
                    Column(
                        horizontalAlignment = Alignment.CenterHorizontally,
                    ) {
                        Text(
                            text = "FocusFlow",
                            color = Color.White,
                            fontSize = 30.sp,
                            fontWeight = FontWeight.Bold,

                            modifier = Modifier.height(8.dp),

        }
    }
}

### One important correction

There is one subtle issue in the above implementation: the shield coordinates are translated **after** they are already scaled. That is harmless visually at normal density but is unnecessary complexity.

I'd actually use this cleaner version of the logo drawing block in production:

```kotlin
Canvas(
    modifier = Modifier.fillMaxSize(),
) {
    val scale = size.minDimension / 132f

    drawCircle(
        color = Color.White.copy(alpha = 0.14f),
        radius = size.minDimension / 2f,
        center = center,
    )

    val left = (size.width - 100f * scale) / 2f
    val top = (size.height - 100f * scale) / 2f

    fun p(x: Float, y: Float) =
        Offset(
            left + x * scale,
            top + y * scale,
        )

    val shield = Path().apply {
        moveTo(p(50f, 22f))
        cubicTo(
            p(42f, 25f),
            p(31f, 28f),
            p(23f, 30f),
        )
        lineTo(p(23f, 50f))
        cubicTo(
            p(23f, 67f),
            p(34f, 82f),
            p(50f, 89f),
        )
        cubicTo(
            p(66f, 82f),
            p(77f, 67f),
            p(77f, 50f),
        )
        lineTo(p(77f, 30f))
        cubicTo(
            p(69f, 28f),
            p(58f, 25f),
            p(50f, 22f),
        )
        close()
    }

    drawPath(
        path = shield,
        color = Color.White,
    )

    val check = Path().apply {
        moveTo(p(36f, 50f))
        lineTo(p(46f, 60f))
        lineTo(p(66f, 38f))
    }

    drawPath(
        path = check,
        color = BrandPrimary,
        style = Stroke(
            width = 5.5f * scale,
            cap = StrokeCap.Round,
            join = StrokeJoin.Round,
        ),
    )
}
```

Use that version inside `FocusFlowSplashLogo()`.
