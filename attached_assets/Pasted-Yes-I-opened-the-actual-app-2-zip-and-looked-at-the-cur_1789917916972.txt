/**
 * The ECG status icon shown in the upper-right of Schedule/Focus screens.
 *
 * This replaces MonitorHeart completely.

@Composable
internal fun ReferenceEcgIcon(
    active: Boolean,
    warning: Boolean = false,
    modifier: Modifier = Modifier,
) {
    val color = when {
        warning -> RefAmber
        active -> RefGreen
        else -> Color(0xFFE2E8F0)
    }

    Canvas(
        modifier = modifier.size(34.dp, 28.dp),
    ) {
        val w = size.width
        val h = size.height

        val path = Path().apply {
            moveTo(w * 0.02f, h * 0.57f)
            lineTo(w * 0.20f, h * 0.57f)
            lineTo(w * 0.31f, h * 0.10f)
            lineTo(w * 0.43f, h * 0.92f)
            lineTo(w * 0.54f, h * 0.40f)
            lineTo(w * 0.62f, h * 0.57f)
            lineTo(w * 0.79f, h * 0.57f)
        }

        drawPath(
            path = path,
            color = color,
            style = Stroke(
                width = 2.4.dp.toPx(),
                cap = StrokeCap.Round,
                join = StrokeJoin.Round,
            ),
        )

        drawCircle(
            color = color,
            radius = h * 0.13f,
            center = Offset(w * 0.88f, h * 0.57f),
            style = Stroke(width = 2.4.dp.toPx()),
        )
    }
}

@Composable
internal fun ReferencePill(
    text: String,
    selected: Boolean,
    modifier: Modifier = Modifier,
    selectedColor: Color = BrandPrimary,
    onClick: (() -> Unit)? = null,
) {
    val shape = CircleShape

    Box(
        modifier = modifier
            .clip(shape)
            .background(
                if (selected) selectedColor
                else RefCard,
            )
            .border(
                width = if (selected) 0.dp else 1.dp,
                color = if (selected) Color.Transparent else RefBorder,
                shape = shape,
            )

                } else {
                    Modifier
                },
            )

    ) {

            color = RefText,
            fontSize = 15.sp,
